controller
import { Controller } from 'egg';

export default class YeDefectNoticeController extends Controller {
  async lotBasicInfo() {
    const { ctx } = this;
    const lotId = String(ctx.query.lotId || '').trim();
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getLotBasicBundle(lotId));
  }

  async dnOptions() {
    const { ctx } = this;
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getDnOptions({
      keyword: String(ctx.query.keyword || '').trim(),
      lotId: String(ctx.query.lotId || '').trim(),
    }));
  }

  async dnDetail() {
    const { ctx } = this;
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getDnDetail(ctx.params.dnId));
  }

  async waferDefectResult() {
    const { ctx } = this;
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getWaferDefectResult({
      lotId: String(ctx.query.lotId || '').trim(),
      waferId: String(ctx.query.waferId || '').trim(),
      recipe: String(ctx.query.recipe || '').trim(),
      defectCode: String(ctx.query.defectCode || '').trim(),
      stepSeq: String(ctx.query.stepSeq || '').trim(),
      stage: String(ctx.query.stage || '').trim(),
      tool: String(ctx.query.tool || '').trim(),
      trackOutTime: String(ctx.query.trackOutTime || '').trim(),
      source: String(ctx.query.source || 'dn').trim(),
      oqa: String(ctx.query.oqa || '').toLowerCase() === 'true',
    }));
  }

  async similarDns() {
    const { ctx } = this;
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getSimilarDns({
      tech: String(ctx.query.tech || '').trim(),
      stage: String(ctx.query.stage || '').trim(),
      defect: String(ctx.query.defect || '').trim(),
      startTime: String(ctx.query.startTime || '').trim(),
      endTime: String(ctx.query.endTime || '').trim(),
      excludeDnId: String(ctx.query.excludeDnId || '').trim(),
    }));
  }

  async suspectSteps() {
    const { ctx } = this;
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getSuspectSteps({
      lotId: String(ctx.query.lotId || '').trim(),
      dnId: String(ctx.query.dnId || '').trim(),
      waferId: String(ctx.query.waferId || '').trim(),
    }));
  }

  async suspectStepDetail() {
    const { ctx } = this;
    ctx.body = this.ok(await ctx.service.yeDefectNotice.getSuspectStepDetail({
      dnId: String(ctx.query.dnId || '').trim(),
      lotId: String(ctx.query.lotId || '').trim(),
      stepSeq: String(ctx.query.stepSeq || '').trim(),
      trackOutTime: String(ctx.query.trackOutTime || '').trim(),
      stepDesc: String(ctx.query.stepDesc || '').trim(),
    }));
  }

  private ok(data: unknown) {
    return { success: true, data };
  }
}



service
import { Service } from 'egg';

type Row = Record<string, any>;

interface DnOptionQuery {
  keyword?: string;
  lotId?: string;
}

interface WaferDefectQuery {
  lotId: string;
  waferId: string;
  recipe: string;
  defectCode: string;
  stepSeq?: string;
  stage?: string;
  tool?: string;
  trackOutTime?: string;
  source?: string;
  oqa?: boolean;
}

interface SimilarDnQuery {
  tech?: string;
  stage: string;
  defect: string;
  startTime: string;
  endTime: string;
  excludeDnId?: string;
}

interface SuspectStepQuery {
  lotId: string;
  dnId?: string;
  waferId?: string;
}

interface SuspectStepDetailQuery {
  dnId?: string;
  lotId: string;
  stepSeq: string;
  trackOutTime?: string;
  stepDesc?: string;
}

export default class YeDefectNoticeService extends Service {
  async getLotBasicBundle(lotId: string) {
    this.assertRequired(lotId, 'lotId is required');

    const [dnId, basicRaw, lotHistory, defectCodes, waferList, modules] = await Promise.all([
      this.getNewDnId(),
      this.getLotBasicInfo(lotId),
      this.getFilteredLotHistory(lotId),
      this.getDefectCodes(),
      this.getWaferList(lotId),
      this.getModuleOptions(),
    ]);

    const basicInfo = await this.decorateLotBasicInfo(basicRaw);
    return {
      dnId,
      basicInfo,
      lotHistory,
      scanSteps: this.getYeScanSteps(lotHistory),
      defectCodes,
      waferList,
      modules,
    };
  }

  async getDnOptions(query: DnOptionQuery) {
    const where: string[] = [ 't.dn_id is not null' ];
    const binds: Row = {};

    if (query.lotId) {
      where.push('t.lotid = :lotId');
      binds.lotId = query.lotId;
    }
    if (query.keyword) {
      where.push(`(
        upper(t.dn_id) like upper(:keyword)
        or upper(t.lotid) like upper(:keyword)
        or upper(t.defectcode) like upper(:keyword)
      )`);
      binds.keyword = `%${query.keyword}%`;
    }

    const rows = await this.queryRows(`
      select *
        from (
          select t.dn_id,
                 t.dn_class,
                 t.lotid,
                 t.defectcode,
                 t.scan_recipe
            from espt_ye_dn_case t
           where ${where.join(' and ')}
           order by t.dn_id desc
        )
       where rownum <= 100
    `, binds);

    return {
      items: rows.map(row => ({
        dnId: row.dn_id,
        dnClass: row.dn_class,
        lotId: row.lotid,
        defectCode: row.defectcode,
        scanRecipe: row.scan_recipe,
        label: `${row.dn_id} ${row.dn_class || ''} ${row.lotid || ''} ${row.defectcode || ''} ${row.scan_recipe || ''}`.trim(),
      })),
    };
  }

  async getDnDetail(dnId: string) {
    this.assertRequired(dnId, 'dnId is required');

    const dnInfo = await this.queryOne(`
      select *
        from espt_ye_dn_case t
       where t.dn_id = :dnId
    `, { dnId });
    this.assertRequired(dnInfo, `DN not found: ${dnId}`);

    const lotId = dnInfo.lotid;
    const [ basicRaw, lotHistory, defectCodes, waferList, modules, defectRows ] = await Promise.all([
      this.getLotBasicInfo(lotId),
      this.getSavedLotHistory(dnId, lotId),
      this.getDefectCodes(),
      this.getWaferList(lotId),
      this.getModuleOptions(),
      this.getSavedDefectSummary(dnId),
    ]);

    return {
      dnId,
      lotId,
      basicInfo: await this.decorateLotBasicInfo({
        ...basicRaw,
        priority: dnInfo.lotpri || basicRaw.priority,
        techg: dnInfo.tech || basicRaw.techg,
        flowName: dnInfo.flowid || basicRaw.flowName,
        lotOwner: dnInfo.lot_owner || basicRaw.lotOwner,
        scraplimit: dnInfo.limit_scrap || basicRaw.scraplimit,
      }),
      lotHistory,
      scanSteps: this.getYeScanSteps(lotHistory),
      scanStepKey: this.findHistoryKey(lotHistory, dnInfo.scan_stepseq, dnInfo.scan_time),
      defectCode: dnInfo.defectcode,
      dnWafer: dnInfo.wafer_ng || dnInfo.wafer_scanned || '',
      defectCodes,
      waferList,
      modules,
      defectResult: defectRows.length ? { summary: this.normalizeDefectSummary(defectRows[0]) } : undefined,
    };
  }

  async getWaferDefectResult(query: WaferDefectQuery) {
    this.assertRequired(query.lotId, 'lotId is required');
    this.assertRequired(query.waferId, 'waferId is required');
    this.assertRequired(query.recipe, 'recipe is required');
    this.assertRequired(query.defectCode, 'defectCode is required');

    const summaryRows = query.oqa
      ? await this.getWaferDefectInfoOqa(query)
      : await this.getWaferDefectInfo(query);

    if (!summaryRows.length) {
      return {
        summary: {},
        dies: [],
        defects: [],
        trend: [],
        pareto: [],
        images: [],
        ng: false,
      };
    }

    const summary = this.normalizeDefectSummary(summaryRows[0]);
    const recipeForMap = query.recipe;
    const inspTool = query.tool || summary.inspect_equip_id || '';
    const inspTime = this.formatOracleDate(summary.inspection_time || query.trackOutTime);

    const [ dies, defects, trend, pareto, images ] = await Promise.all([
      this.getAllDieCoordinates(query.lotId, query.waferId, recipeForMap, inspTool, inspTime),
      this.getAllDefectCoordinates(query.lotId, query.waferId, recipeForMap, inspTool, inspTime, query.defectCode),
      this.getDefectTrend(query.recipe, query.defectCode, inspTool, [ `${this.baseLot(query.lotId)}#${query.waferId}` ]),
      this.getDefectPareto(query.lotId, query.waferId, recipeForMap, inspTool, inspTime),
      this.getDefectImages(query.lotId, query.waferId, recipeForMap, inspTool, inspTime, query.defectCode),
    ]);

    return {
      summary,
      dies,
      defects,
      trend,
      pareto,
      images,
      ng: Number(summary.dc || 0) > 0,
      source: query.source || 'dn',
    };
  }

  async getSimilarDns(query: SimilarDnQuery) {
    this.assertRequired(query.stage, 'stage is required');
    this.assertRequired(query.defect, 'defect is required');
    this.assertRequired(query.startTime, 'startTime is required');
    this.assertRequired(query.endTime, 'endTime is required');

    const stageCondition = /^[MT]/i.test(query.stage)
      ? `t.scan_stage like :stage`
      : `t.scan_stage = :stage`;
    const stageBind = /^[MT]/i.test(query.stage)
      ? `${query.stage.substring(0, 1)}_${query.stage.substring(2)}`
      : query.stage;

    const rows = await this.queryRows(`
      select t.dn_id,
             t.dn_class,
             t.lotid,
             t.product,
             t.tech,
             t.scan_stage,
             t.scan_recipe,
             t.scan_machine,
             t.defectcode,
             t.dn_status
        from espt_ye_dn_case t
       where ${stageCondition}
         and t.defectcode = :defect
         and (:excludeDnId is null or t.dn_id <> :excludeDnId)
         and t.dn_id in (
           select distinct a.dn_no
             from espt_ye_dn_action_log_temp a
            where a.action like 'Initial DN%'
              and a.action_time between to_date(:startTime, 'yyyy-mm-dd')
                                  and to_date(:endTime, 'yyyy-mm-dd') + 1 - 1 / 1440
         )
       order by t.dn_id
    `, {
      stage: stageBind,
      defect: query.defect,
      excludeDnId: query.excludeDnId || null,
      startTime: query.startTime,
      endTime: query.endTime,
    });

    return { items: rows };
  }

  async getSuspectSteps(query: SuspectStepQuery) {
    this.assertRequired(query.lotId, 'lotId is required');

    const [ lotHistory, usedSteps, modules ] = await Promise.all([
      this.getFilteredLotHistory(query.lotId, query.waferId),
      query.dnId ? this.getExistingSuspectStepKeys(query.dnId) : Promise.resolve(new Set<string>()),
      this.getModuleOptions(),
    ]);

    const items = lotHistory
      .filter(step => !usedSteps.has(`${step.stepseq || ''}__${step.trackouttime || ''}__${step.stepdesc || 'Null'}`))
      .map(step => ({
        ...step,
        suspectKey: `${step.stepseq || ''}__${step.trackouttime || ''}__${step.stepdesc || 'Null'}`,
        label: `${step.stepseq || ''}__${step.trackouttime || ''}__${step.stepdesc || 'Null'}`,
        informedModule: this.guessModule(step),
      }));

    return { items, modules };
  }

  async getSuspectStepDetail(query: SuspectStepDetailQuery) {
    this.assertRequired(query.lotId, 'lotId is required');
    this.assertRequired(query.stepSeq, 'stepSeq is required');

    const lotHistory = await this.getFilteredLotHistory(query.lotId);
    const detail = lotHistory.find(step => {
      const desc = step.stepdesc || 'Null';
      return String(step.stepseq || '') === query.stepSeq
        && (!query.trackOutTime || String(step.trackouttime || '') === query.trackOutTime)
        && (!query.stepDesc || desc === query.stepDesc);
    });

    if (!detail) return { detail: {}, modules: await this.getModuleOptions() };

    const [ ngWafers, stepWafers, modules ] = await Promise.all([
      query.dnId ? this.getNgWafers(query.dnId) : Promise.resolve([]),
      this.getStepWaferList(query.lotId, query.stepSeq),
      this.getModuleOptions(),
    ]);

    const missing = stepWafers.length ? ngWafers.filter(wf => !stepWafers.includes(wf)) : [];
    const warning = stepWafers.length === 0
      ? 'No wafer list was found for this process step. Please confirm the selected suspect step.'
      : missing.length
        ? `${missing.join(',')} not found in process step wafer list ${stepWafers.join(',')}. Please confirm the selected suspect step.`
        : '';

    return {
      detail: {
        ...detail,
        informedModule: this.guessModule(detail),
      },
      modules,
      warning,
    };
  }

  private async getLotBasicInfo(lotId: string) {
    const row = await this.queryOne(`
      select t.lotname,
             t.productname as productspecname,
             t.priority,
             substr(t.productname, 1, 4) as prodbody,
             t.productflowname as processflowname,
             t.productquantity,
             t.lotowner
        from rpt.base_lot t
       where t.lotname = :lotId
    `, { lotId });
    this.assertRequired(row, `Lot not found: ${lotId}`);
    return row;
  }

  private async decorateLotBasicInfo(basic: Row) {
    if (!basic) return {};

    const [ flowName, lotOwner, maskInfo ] = await Promise.all([
      basic.flowName ? Promise.resolve(basic.flowName) : this.getFlowName(basic.productspecname),
      basic.lotOwner ? Promise.resolve(basic.lotOwner) : this.getUserLabel(basic.lotowner),
      basic.techg ? Promise.resolve(basic) : this.getLotMaskInfo(basic.prodbody),
    ]);

    return {
      ...basic,
      flowName: flowName || basic.processflowname || '',
      lotOwner: lotOwner || basic.lotowner || '',
      techg: maskInfo.techg || basic.techg || '',
      shotSize: String(maskInfo.mask || '').split('*')[0] || basic.shotSize || '',
      scraplimit: maskInfo.scraplimit || basic.scraplimit || '',
      yesponsor: maskInfo.yesponsor || basic.yesponsor || '',
    };
  }

  private async getFilteredLotHistory(lotId: string, waferId = '') {
    const baseLot = this.baseLot(lotId);
    const [ lotHistory, stepTimeInfo, chamberHistory ] = await Promise.all([
      this.getLotHistoryNew(baseLot),
      this.getStepHistoryNew(baseLot),
      waferId ? this.getChamberHistoryNew(baseLot, waferId) : Promise.resolve([]),
    ]);

    const timeRows = chamberHistory.length
      ? stepTimeInfo.concat(chamberHistory).filter(row => row.flag === 'key')
      : stepTimeInfo;

    const timeMap = new Map<string, Row>();
    for (const row of timeRows) {
      timeMap.set(this.historyMatchKey(row), row);
    }

    const merged = lotHistory
      .map(row => {
        const matched = timeMap.get(this.historyMatchKey(row));
        const normalized = {
          ...row,
          stepseq: row.stepseq || row.stepsequence || '',
          machinename: row.machinename || row.machine || '',
          productspecname: row.productspecname || row.product || '',
          trackintime: matched?.trackintime || row.trackintime || '',
        };
        return this.decorateHistoryRow(normalized);
      })
      .filter(row => row.trackintime);

    return merged
      .sort((a, b) => {
        const timeCompare = String(b.trackintime || '').localeCompare(String(a.trackintime || ''));
        if (timeCompare) return timeCompare;
        return String(b.stepseq || '').localeCompare(String(a.stepseq || ''));
      })
      .map((row, index) => ({
        ...row,
        historyKey: `${row.stepseq || ''}_${row.trackouttime || ''}_${index}`,
      }));
  }

  private async getLotHistoryNew(lotId: string) {
    return await this.queryRows(`
      select a.lotname,
             a.productqty,
             a.priority,
             a.stage,
             a.stepsequence,
             a.stepsequence as stepseq,
             a.operation,
             a.stepdescription as stepdesc,
             a.machinename,
             a.recipe,
             a.trackintime,
             a.trackouttime,
             a.event,
             a.productname as productspecname,
             a.lottype
        from rpt.lot_step_history a
       where a.lotname = :lotId
       order by a.trackouttime desc, a.stepsequence desc
    `, { lotId });
  }

  private async getStepHistoryNew(lotId: string) {
    return await this.queryRows(`
      select a.lotname,
             a.stepsequence,
             a.stepsequence as stepseq,
             a.stage,
             a.event,
             a.machinename,
             a.trackintime,
             a.trackouttime,
             'key' as flag
        from rpt.lot_step_history a
       where a.lotname = :lotId
         and a.trackintime is not null
    `, { lotId });
  }

  private async getChamberHistoryNew(lotId: string, waferId: string) {
    return await this.queryRows(`
      select y.lotid as lotname,
             y.currentflowseq as stepsequence,
             y.currentflowseq as stepseq,
             y.stage,
             y.event,
             y.machinename,
             y.trackintime,
             y.trackouttime,
             'key' as flag
        from rpt.chamber_history y
       where y.lotid = :lotId
         and substr(y.waferid, instr(y.waferid, '#') + 1) = :waferId
    `, { lotId, waferId: waferId.replace(/^#/, '') });
  }

  private async getSavedLotHistory(dnId: string, fallbackLotId: string) {
    const rows = await this.queryRows(`
      select case when t.currentsite is not null then t.currentsite else 'FAB6' end as currentsite,
             t.lot as lotname,
             t.qty as productqty,
             t.priority,
             t.stage,
             t.stepseq,
             t.operation,
             t.stepdesc,
             t.machine as machinename,
             t.recipe,
             to_char(to_date(t.trackintime, 'yyyy/mm/dd hh24:mi:ss'), 'yyyy/mm/dd hh24:mi:ss') as trackintime,
             to_char(to_date(t.trackouttime, 'yyyy/mm/dd hh24:mi:ss'), 'yyyy/mm/dd hh24:mi:ss') as trackouttime,
             t.event,
             t.product as productspecname,
             t.lottype
        from espt_ye_dn_lot_history_temp t
       where t.dn_id = :dnId
       order by to_date(t.trackintime, 'yyyy/mm/dd hh24:mi:ss') desc, t.stepseq desc
    `, { dnId });

    if (rows.length) {
      return rows.map((row, index) => ({ ...row, historyKey: `${row.stepseq || ''}_${row.trackouttime || ''}_${index}` }));
    }
    return await this.getFilteredLotHistory(fallbackLotId);
  }

  private async getWaferDefectInfo(query: WaferDefectQuery) {
    const { product, layerId } = this.parseRecipe(query.recipe);
    const rows = await this.queryRows(`
      select tt.*,
             decode(tt.reviewDieCnt, 0, 0,
                    round(((tt.defectDieCnt / tt.reviewDieCnt) * tt.totalDieCnt) / tt.inspected_die, 4)) as ndd_new
        from (
          select distinct t.layer_id,
                 t.wafer_key,
                 t.inspection_time,
                 t.inspected_die,
                 t.inspect_equip_id,
                 t.defects as tc,
                 (select count(a.class_number)
                    from insp_defect a
                   where a.inspection_time = t.inspection_time
                     and t.wafer_key = a.wafer_key
                     and a.class_number <> '0') as rdc,
                 (select count(a.class_number)
                    from insp_defect a
                   where a.inspection_time = t.inspection_time
                     and t.wafer_key = a.wafer_key
                     and a.class_number = :defectCode) as dc,
                 t.adder_random_defects as ac,
                 round((select count(distinct a.index_x || '#' || a.index_y)
                          from insp_defect a
                         where a.inspection_time = t.inspection_time
                           and t.wafer_key = a.wafer_key
                           and a.class_number = :defectCode) / t.inspected_die, 4) as ndd,
                 (select count(distinct b.index_x || '_' || b.index_y)
                    from insp_defect b
                   where b.inspection_time = t.inspection_time
                     and t.wafer_key = b.wafer_key) as totalDieCnt,
                 (select count(distinct b.index_x || '_' || b.index_y)
                    from insp_defect b
                   where b.inspection_time = t.inspection_time
                     and t.wafer_key = b.wafer_key
                     and b.class_number = :defectCode) as defectDieCnt,
                 (select count(distinct b.index_x || '_' || b.index_y)
                    from insp_defect b
                   where b.inspection_time = t.inspection_time
                     and t.wafer_key = b.wafer_key
                     and b.class_number <> '0') as reviewDieCnt
            from insp_wafer_summary t,
                 insp_defect b
           where t.lot_id = :lotId
             and t.wafer_id = :waferId
             and b.inspection_time = t.inspection_time
             and t.wafer_key = b.wafer_key
             and t.layer_id in (:layerId, :layerIdT)
        ) tt
       order by tt.inspection_time desc
    `, {
      lotId: query.lotId,
      waferId: query.waferId,
      defectCode: query.defectCode,
      layerId,
      layerIdT: `${layerId}_T`,
    });

    if (!(await this.isHGroupProduct(product)) || !rows.length) return rows;
    return rows;
  }

  private async getWaferDefectInfoOqa(query: WaferDefectQuery) {
    return await this.queryRows(`
      select tt.*,
             decode(tt.reviewDieCnt, 0, 0,
                    round(((tt.defectDieCnt / tt.reviewDieCnt) * tt.totalDieCnt) / tt.inspected_die, 4)) as ndd_new
        from (
          select distinct t.layer_id,
                 t.wafer_key,
                 t.inspection_time,
                 t.inspected_die,
                 t.inspect_equip_id,
                 t.defects as tc,
                 (select count(a.class_number)
                    from insp_defect a
                   where a.inspection_time = t.inspection_time
                     and t.wafer_key = a.wafer_key
                     and a.class_number <> '0') as rdc,
                 (select count(a.class_number)
                    from insp_defect a
                   where a.inspection_time = t.inspection_time
                     and t.wafer_key = a.wafer_key
                     and a.class_number = :defectCode) as dc,
                 t.adder_random_defects as ac,
                 round(t.defective_die / t.inspected_die, 4) as ndd,
                 (select count(distinct b.index_x || '_' || b.index_y)
                    from insp_defect b
                   where b.inspection_time = t.inspection_time
                     and t.wafer_key = b.wafer_key) as totalDieCnt,
                 (select count(distinct b.index_x || '_' || b.index_y)
                    from insp_defect b
                   where b.inspection_time = t.inspection_time
                     and t.wafer_key = b.wafer_key
                     and b.class_number = :defectCode) as defectDieCnt,
                 (select count(distinct b.index_x || '_' || b.index_y)
                    from insp_defect b
                   where b.inspection_time = t.inspection_time
                     and t.wafer_key = b.wafer_key
                     and b.class_number <> '0') as reviewDieCnt
            from insp_wafer_summary t,
                 insp_defect b,
                 insp_recipe e
           where t.lot_id = :lotId
             and t.wafer_id = :waferId
             and b.inspection_time = t.inspection_time
             and t.wafer_key = b.wafer_key
             and t.recipe_key = e.recipe_key
             and e.recipe_id = :recipe
        ) tt
       order by tt.inspection_time desc
    `, {
      lotId: query.lotId,
      waferId: query.waferId,
      defectCode: query.defectCode,
      recipe: query.recipe,
    });
  }

  private normalizeDefectSummary(row: Row) {
    const tc = Number(row.tc ?? row.TC ?? 0);
    const rdc = Number(row.rdc ?? row.RDC ?? 0);
    const dc = Number(row.dc ?? row.DC ?? 0);
    const ac = Number(row.ac ?? row.AC ?? 0);
    const ndd = Number(row.ndd ?? row.NDD ?? 0);
    const nddNew = Number(row.ndd_new ?? row.nddNew ?? row.NDD_NEW ?? 0);
    const ndc = ac >= rdc ? (rdc === 0 ? '' : Math.round((dc * ac) / rdc)) : dc;

    return {
      ...row,
      tc,
      rdc,
      dc,
      ac,
      ndd,
      nddNew,
      ndc,
      nddPercent: Number.isFinite(ndd) ? (ndd * 100).toFixed(2) : '',
      nddNewPercent: Number.isFinite(nddNew) ? (nddNew * 100).toFixed(2) : '',
    };
  }

  private async getAllDefectCoordinates(
    lotId: string,
    waferId: string,
    recipe: string,
    inspTool: string,
    inspTime: string,
    defectCode: string,
  ) {
    if (!inspTool || !inspTime) return [];
    const rows = await this.queryRows(`
      select d.index_x - r.origin_index_x as die_x,
             d.index_y - r.origin_index_y as die_y,
             (d.wafer_x - d.index_x * r.die_size_x) / 1000.0 as defect_in_die_x,
             (d.wafer_y - d.index_y * r.die_size_y) / 1000.0 as defect_in_die_y,
             r.die_size_x / 1000 as die_size_x,
             r.die_size_y / 1000 as die_size_y,
             a.name || '_' || a.code as defect_class,
             w.inspection_time,
             w.inspect_equip_id
        from insp_defect d,
             insp_wafer_summary w,
             insp_recipe r,
             (
               select distinct h.code, h.name
                 from insp_decode h
                where h.parm_key = 2
             ) a
       where d.inspection_time = w.inspection_time
         and d.wafer_key = w.wafer_key
         and w.recipe_key = r.recipe_key
         and d.class_number = a.code
         and w.lot_id = :lotId
         and w.wafer_id = :waferId
         and r.recipe_id like :recipe
         and w.inspect_equip_id = :inspTool
         and w.inspection_time = to_date(:inspTime, 'yyyy/mm/dd hh24:mi:ss')
         and (:defectCode is null or d.class_number = :defectCode or a.name || '_' || a.code = :defectCode)
    `, {
      lotId,
      waferId,
      recipe: `${recipe}%`,
      inspTool,
      inspTime,
      defectCode: defectCode || null,
    });

    return rows.map(row => ({
      index_x: Number(row.die_x || 0) + Number(row.defect_in_die_x || 0) / Math.max(Number(row.die_size_x || 1), 1),
      index_y: Number(row.die_y || 0) + Number(row.defect_in_die_y || 0) / Math.max(Number(row.die_size_y || 1), 1),
      defect_class: row.defect_class,
      time: row.inspection_time,
      tool: row.inspect_equip_id,
    }));
  }

  private async getDefectTrend(recipe: string, defectCode: string, machine: string, dnWafers: string[]) {
    const rows = await this.queryRows(`

        from (

  private guessModule(step: Row) {
    const text = `${step.oldmachinecapability || ''} ${step.operation || ''} ${step.machinename || ''}`.toUpperCase();
    if (text.includes('L-ADI') || text.includes('L-AEI') || text.includes('LITHO')) return 'Litho';
    if (text.includes('ETCH')) return 'Etch';
    if (text.includes('CMP')) return 'CMP';
    if (text.includes('CVD') || text.includes('PVD') || text.includes('TF')) return 'ThinFilm';
    if (text.includes('IMP')) return 'IMP';
    return '';
  }

  private baseLot(lotId: string) {
    return String(lotId || '').split('.')[0];
  }

  private todayYmd() {
    const date = new Date();
    const y = date.getFullYear();
    const m = String(date.getMonth() + 1).padStart(2, '0');
    const d = String(date.getDate()).padStart(2, '0');
    return `${y}${m}${d}`;
  }

  private formatOracleDate(value: any) {
    if (!value) return '';
    if (value instanceof Date) {
      const y = value.getFullYear();
      const m = String(value.getMonth() + 1).padStart(2, '0');
      const d = String(value.getDate()).padStart(2, '0');
      const hh = String(value.getHours()).padStart(2, '0');
      const mm = String(value.getMinutes()).padStart(2, '0');
      const ss = String(value.getSeconds()).padStart(2, '0');
      return `${y}/${m}/${d} ${hh}:${mm}:${ss}`;
    }
    return String(value).replace(/-/g, '/');
  }

  private assertRequired<T>(value: T, message: string): asserts value is NonNullable<T> {
    if (value === undefined || value === null || value === '') {
      this.ctx.throw(400, message);
    }
  }

  private async queryOne<T extends Row = Row>(sql: string, binds: Row = {}) {
    const rows = await this.queryRows<T>(sql, binds);
    return rows[0];
  }

  private async queryRows<T extends Row = Row>(sql: string, binds: Row = {}): Promise<T[]> {
    const client = (this.app as any).oracle || (this.app as any).oracledb || (this.ctx as any).oracle;
    if (!client) {
      this.ctx.throw(500, 'Oracle client is not configured. Expected app.oracle/app.oracledb/ctx.oracle.');
    }

    const executor = client.execute || client.query;
    if (!executor) {
      this.ctx.throw(500, 'Oracle client must expose execute(sql, binds) or query(sql, binds).');
    }

    const result = await executor.call(client, sql, binds);
    const rows = Array.isArray(result) ? result : result.rows || [];
    const meta = result.metaData || result.metadata || [];

    return rows.map((row: any) => {
      if (!Array.isArray(row)) return this.lowerKeys(row) as T;
      const obj: Row = {};
      meta.forEach((item: any, index: number) => {
        obj[String(item.name || item).toLowerCase()] = row[index];
      });
      return obj as T;
    });
  }

  private lowerKeys(row: Row) {
    const obj: Row = {};
    Object.entries(row || {}).forEach(([ key, value ]) => {
      obj[key.toLowerCase()] = value;
    });
    return obj;
  }

async function getLotBasicInfo() {
  if (!form.lotId) {
    ElMessage.warning("请输入 Lot ID");
    return;

  const payload = await apiGet("/lots/basic-info", { lotId: form.lotId });
  setObject(lotBasic, payload.basicInfo || {});
  form.dnNo = payload.dnId || "";
  lotHistory.value = normalizeHistory(payload.lotHistory || []);
  scanStepOptions.value = normalizeStepOptions(payload.scanSteps, lotHistory.value);
  defectCodeOptions.value = payload.defectCodes || defectCodeOptions.value;
  waferOptions.value = payload.waferList || waferOptions.value;
  moduleOptions.value = payload.modules || moduleOptions.value;

  if (!lotHistory.value.length) {
    ElMessage.warning("未查询到 Lot History");
    return;
  }
  if (scanStepOptions.value.length === 1) {
    form.scanStepKey = scanStepOptions.value[0].key;
    handleScanStepChange(form.scanStepKey);
  }
}

async function getExistDnInfo() {
  if (!form.dnNo) {
    ElMessage.warning("请选择 DN No");
    return;

  const payload = await apiGet(`/dns/${encodeURIComponent(form.dnNo)}`);
  setObject(lotBasic, payload.basicInfo || {});
  form.lotId = payload.lotId || lotBasic.lotname || form.lotId;
  form.defectCode = payload.defectCode || "";
  form.scanStepKey = payload.scanStepKey || "";
  form.dnWafer = payload.dnWafer || "";
  lotHistory.value = normalizeHistory(payload.lotHistory || []);
  scanStepOptions.value = normalizeStepOptions(payload.scanSteps, lotHistory.value);
  defectCodeOptions.value = payload.defectCodes || defectCodeOptions.value;
  waferOptions.value = payload.waferList || waferOptions.value;
  moduleOptions.value = payload.modules || moduleOptions.value;
  handleScanStepChange(form.scanStepKey);

  if (payload.defectResult) {
    defectResult.dn = normalizeDefectResult(payload.defectResult);
  }
}

async function loadWaferDefect(mode) {
  const wafer = mode === "add" ? form.addScanWafer : form.dnWafer;
  const scanStep = mode === "pre" ? findHistoryByKey(form.preScanStepKey) : selectedScanStep;

  if (!form.lotId || !wafer || !form.defectCode || !scanStep.recipe) {
    ElMessage.warning("Lot、Wafer、Defect Type 和 Scan Step 必须完整");
    return;
  }

  loading.defect = true;
  try {
    const payload = await apiGet("/wafers/defect-result", {
      lotId: form.lotId,
      waferId: wafer.replace(/^#/, ""),
      defectCode: lastSegment(form.defectCode),
      recipe: scanStep.recipe,
      stepSeq: scanStep.stepseq,
      stage: scanStep.stage,
      tool: scanStep.machinename,
      trackOutTime: scanStep.trackouttime,
      source: mode,
      oqa: String(scanStep.recipe || "").startsWith("AVI"),
    });

    defectResult[mode] = normalizeDefectResult(payload);
    defectResult.activeMode = mode;
    visualTab.value = "map";
  } catch (error) {
    showApiError(error, "Defect 信息获取失败");
  } finally {
    loading.defect = false;
  }
}

async function apiGet(path, params = {}) {
  const url = new URL(`${API_BASE}${path}`, window.location.origin);
  Object.entries(params).forEach(([key, value]) => {
    if (value !== undefined && value !== null && value !== "") url.searchParams.set(key, value);
  });

  const response = await fetch(url.toString(), {
    headers: { Accept: "application/json" },
    credentials: "include",
  });

  if (!response.ok) {
    throw new Error(`${response.status} ${response.statusText}`);
  }

  const body = await response.json();
  if (body.success === false) throw new Error(body.message || "API failed");
  return body.data ?? body;
}

function normalizeHistory(rows) {
  return rows.map((row, index) => ({
    ...row,
    historyKey: row.historyKey || `${row.stepseq || row.stepsequence || ""}_${row.trackouttime || ""}_${index}`,
    stepseq: row.stepseq || row.stepsequence || "",
    machinename: row.machinename || row.machine || "",
    productspecname: row.productspecname || row.product || "",
  }));
}

function buildScanStepOptions(rows) {
  return rows
    .filter((row) => /^(BIYOCAP|BYY|SRC)/i.test(row.operation || ""))
    .sort((a, b) => String(b.trackouttime || "").localeCompare(String(a.trackouttime || "")))
    .map((row) => ({ key: row.historyKey, label: formatStepLabel(row), raw: row }));
}

function normalizeStepOptions(serverOptions, rows) {
  if (!Array.isArray(serverOptions) || !serverOptions.length) return buildScanStepOptions(rows);
  const rowByKey = new Map(rows.map((row) => [row.historyKey, row]));
  return serverOptions.map((item) => ({
    key: item.key,
    label: item.label || formatStepLabel(item.raw || rowByKey.get(item.key) || {}),
    raw: item.raw || rowByKey.get(item.key) || {},
  }));
}

function normalizeDefectResult(payload) {
  return {
    summary: payload.summary || payload.defectInfo || {},
    dies: payload.dies || payload.dieCoordinates || [],
    defects: payload.defects || payload.defectCoordinates || [],
    trend: payload.trend || [],
    pareto: payload.pareto || [],
    images: payload.images || [],
    ng: Boolean(payload.ng),
  };
}

function setObject(target, source) {
  Object.keys(target).forEach((key) => delete target[key]);
  Object.assign(target, source || {});
}

function findHistoryByKey(key) {
  return lotHistory.value.find((item) => item.historyKey === key) || {};
}

function formatStepLabel(row) {
  return `${row.stepseq || ""}_${row.trackouttime || ""}_${row.stage || ""}`;
}

function normalizeTime(value) {
  if (!value) return 0;
  const timestamp = new Date(String(value).replace(/-/g, "/")).getTime();
  return Number.isNaN(timestamp) ? 0 : timestamp;
}

function lastSegment(value) {
  return String(value || "").split("_").filter(Boolean).pop() || "";
}

function valueOrDash(value) {
  return value === undefined || value === null || value === "" ? "-" : value;
}

function toPercent(value) {
  if (value === undefined || value === null || value === "") return "-";
  const number = Number(value);
  return Number.isFinite(number) ? (number * 100).toFixed(2) : "-";
}

function calcNdc(summary) {
  const dc = Number(summary.DC ?? summary.dc);
  const ac = Number(summary.AC ?? summary.ac);
  const rdc = Number(summary.RDC ?? summary.rdc);
  if (!Number.isFinite(dc) || !Number.isFinite(ac) || !Number.isFinite(rdc)) return "";
  if (ac >= rdc) return rdc === 0 ? "" : Math.round((dc * ac) / rdc);
  return dc;
}

function guessModule(step) {
  const capability = `${step.oldmachinecapability || ""} ${step.operation || ""}`.toUpperCase();
  if (capability.includes("L-ADI") || capability.includes("L-AEI")) return "Litho";
  if (capability.includes("ETCH")) return "Etch";
  if (capability.includes("CMP")) return "CMP";
  if (capability.includes("CVD") || capability.includes("PVD")) return "ThinFilm";
  return "";
}

function defaultSimilarRange() {
  const end = new Date();
  const start = new Date();
  start.setDate(end.getDate() - 90);
  return [formatDate(start), formatDate(end)];
}

function formatDate(date) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

function showApiError(error, fallback) {
  ElMessage.error(`${fallback}: ${error.message || error}`);
}

const WaferMapSvg = defineComponent({
  name: "WaferMapSvg",
  props: {
    dies: { type: Array, default: () => [] },
    defects: { type: Array, default: () => [] },
    ng: { type: Boolean, default: false },
  },
  setup(props) {
    const bounds = computed(() => {
      const xs = props.dies.map((item) => Number(item.index_x ?? item.x)).filter(Number.isFinite);
      const ys = props.dies.map((item) => Number(item.index_y ?? item.y)).filter(Number.isFinite);
      return {
        minX: Math.min(...xs, 0),
        maxX: Math.max(...xs, 1),
        minY: Math.min(...ys, 0),
        maxY: Math.max(...ys, 1),
      };
    });

    const project = (x, y) => {
      const b = bounds.value;
      const w = Math.max(b.maxX - b.minX + 1, 1);
      const hgt = Math.max(b.maxY - b.minY + 1, 1);
      return {
        x: 20 + ((Number(x) - b.minX) / w) * 320,
        y: 20 + ((b.maxY - Number(y)) / hgt) * 320,
        sizeX: Math.max(320 / w - 1, 2),
        sizeY: Math.max(320 / hgt - 1, 2),
      };
    };

    return () =>
      h("svg", { viewBox: "0 0 360 380", class: "wafer-svg", role: "img" }, [
        h("rect", { x: 0, y: 0, width: 360, height: 380, rx: 6, fill: props.ng ? "#fff3d4" : "#ffffff" }),
        h("circle", { cx: 180, cy: 180, r: 168, fill: "#f8fafc", stroke: "#334155", "stroke-width": 2 }),
        props.dies.map((die) => {
          const p = project(die.index_x ?? die.x, die.index_y ?? die.y);
          return h("rect", {
            x: p.x,
            y: p.y,
            width: p.sizeX,
            height: p.sizeY,
            fill: "#e2e8f0",
            stroke: "#94a3b8",
            "stroke-width": 0.4,
          });
        }),
        props.defects.map((defect) => {
          const p = project(defect.index_x ?? defect.x ?? defect.die_x, defect.index_y ?? defect.y ?? defect.die_y);
          return h("circle", {
            cx: p.x + p.sizeX / 2,
            cy: p.y + p.sizeY / 2,
            r: 3,
            fill: defect.defect_class ? "#ef4444" : "#f97316",
            opacity: 0.88,
          });
        }),
        h("text", { x: 180, y: 365, "text-anchor": "middle", class: "svg-caption" }, "Wafer Defect Map"),
      ]);
  },
});

const ParetoChartSvg = defineComponent({

          const height = (value / max) * 220;
          return h("g", {}, [
            h("rect", { x, y: 286 - height, width: barWidth, height, fill: "#2563eb", rx: 2 }),
            h("text", { x: x + barWidth / 2, y: 280 - height, "text-anchor": "middle", class: "svg-value" }, value),
            h("text", { x: x + barWidth / 2, y: 306, "text-anchor": "end", transform: `rotate(-45 ${x + barWidth / 2} 306)`, class: "svg-label" }, item.name || item.label || item.defect_class || index + 1),
          ]);
        }),
        h("text", { x: 340, y: 22, "text-anchor": "middle", class: "svg-caption" }, "Defect Pareto"),
      ]);
    };
  },
});

function renderLineChart(points, title) {
  const items = points.slice(-100);
  const values = items.map((item) => Number(item.value ?? item.tc ?? item.dc ?? item.ndc)).filter(Number.isFinite);
  const max = Math.max(...values, 1);
  const min = Math.min(...values, 0);
  const span = Math.max(max - min, 1);
  const coords = items.map((item, index) => {
    const value = Number(item.value ?? item.tc ?? item.dc ?? item.ndc) || 0;
    const x = 50 + (index / Math.max(items.length - 1, 1)) * 590;
    const y = 286 - ((value - min) / span) * 230;
    return { x, y, value, key: item.key || `${item.lot_id || ""}#${item.wafer_id || ""}` };
  });
  const path = coords.map((point, index) => `${index ? "L" : "M"} ${point.x} ${point.y}`).join(" ");

  return h("svg", { viewBox: "0 0 680 340", class: "chart-svg", role: "img" }, [
    h("rect", { width: 680, height: 340, rx: 6, fill: "#ffffff" }),
    h("line", { x1: 46, y1: 286, x2: 640, y2: 286, stroke: "#334155" }),
    h("line", { x1: 46, y1: 30, x2: 46, y2: 286, stroke: "#334155" }),
    h("path", { d: path, fill: "none", stroke: "#0f766e", "stroke-width": 2.5 }),
    coords.map((point) =>
      h("circle", {
        cx: point.x,
        cy: point.y,
        r: 4,
        fill: point.key && point.key.includes("#") ? "#dc2626" : "#0f766e",
      }),
    ),
    h("text", { x: 340, y: 22, "text-anchor": "middle", class: "svg-caption" }, title),
  ]);
}
